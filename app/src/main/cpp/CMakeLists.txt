cmake_minimum_required(VERSION 3.22.1)
project(word_diffusion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# ggml's matrix kernels are C as well as C++; optimize both in debug APKs.
add_compile_options(-O3)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
option(WORD_DIFFUSION_VULKAN "Build the experimental image benchmark GPU backend" ON)
if(WORD_DIFFUSION_VULKAN)
    include(FetchContent)
    FetchContent_Declare(word_vulkan_headers
        GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git
        GIT_TAG 19725e4d48082fe78e26622b15d3080ccd54112b)
    FetchContent_Declare(word_spirv_headers
        GIT_REPOSITORY https://github.com/KhronosGroup/SPIRV-Headers.git
        GIT_TAG 01e0577914a75a2569c846778c2f93aa8e6feddd)
    FetchContent_GetProperties(word_vulkan_headers)
    if(NOT word_vulkan_headers_POPULATED)
        FetchContent_Populate(word_vulkan_headers)
    endif()
    FetchContent_GetProperties(word_spirv_headers)
    if(NOT word_spirv_headers_POPULATED)
        FetchContent_Populate(word_spirv_headers)
    endif()
    set(Vulkan_INCLUDE_DIR "${word_vulkan_headers_SOURCE_DIR}/include" CACHE PATH "" FORCE)
    set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/${ANDROID_HOST_TAG}/glslc" CACHE FILEPATH "")
    # ggml needs the header-only SPIR-V target; no host installation is required.
    set(_spirv_config_dir "${CMAKE_CURRENT_BINARY_DIR}/spirv-config")
    file(MAKE_DIRECTORY "${_spirv_config_dir}")
    file(WRITE "${_spirv_config_dir}/SPIRV-HeadersConfig.cmake"
        "if(NOT TARGET SPIRV-Headers::SPIRV-Headers)\nadd_library(SPIRV-Headers::SPIRV-Headers INTERFACE IMPORTED)\nset_target_properties(SPIRV-Headers::SPIRV-Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES \"${word_spirv_headers_SOURCE_DIR}/include\")\nendif()\n")
    set(SPIRV-Headers_DIR "${_spirv_config_dir}" CACHE PATH "" FORCE)
    unset(_word_host_cc CACHE)
    unset(_word_host_cxx CACHE)
    find_program(_word_host_cc NAMES clang gcc PATHS /usr/bin NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_program(_word_host_cxx NAMES clang++ g++ PATHS /usr/bin NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    set(_word_host_toolchain "${CMAKE_CURRENT_BINARY_DIR}/word-shader-host.cmake")
    file(WRITE "${_word_host_toolchain}"
        "set(CMAKE_SYSTEM_NAME ${CMAKE_HOST_SYSTEM_NAME})\nset(CMAKE_C_COMPILER \"${_word_host_cc}\")\nset(CMAKE_CXX_COMPILER \"${_word_host_cxx}\")\nset(CMAKE_MAKE_PROGRAM \"${CMAKE_MAKE_PROGRAM}\" CACHE FILEPATH \"\" FORCE)\nset(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)\nset(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY NEVER)\nset(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE NEVER)\n")
    set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${_word_host_toolchain}" CACHE FILEPATH "" FORCE)
    set(SD_VULKAN ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(../../../../third_party/stable-diffusion.cpp ${CMAKE_CURRENT_BINARY_DIR}/sdcpp)
if(WORD_DIFFUSION_VULKAN)
    target_include_directories(ggml-vulkan PRIVATE "${word_spirv_headers_SOURCE_DIR}/include")
endif()
add_library(word_diffusion SHARED diffusion_jni.cpp)
target_link_libraries(word_diffusion PRIVATE stable-diffusion log)
if(WORD_DIFFUSION_VULKAN)
    target_compile_definitions(word_diffusion PRIVATE WORD_DIFFUSION_VULKAN=1)
endif()
target_link_options(word_diffusion PRIVATE "-Wl,-z,max-page-size=16384")
